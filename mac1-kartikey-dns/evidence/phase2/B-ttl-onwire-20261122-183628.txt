# ==== Phase 2 / Extension B -- short visible TTL ====
# Installed as /opt/homebrew/etc/dnsmasq.d/ttl.conf
# dnsmasq's default TTL for locally configured records is 0 (= "do not cache"),
# which is why Phase 1 record changes were visible instantly. 30s makes caching observable.
local-ttl=30

TTL 0  : clients may NOT cache -> every lookup hits the server, changes visible instantly
TTL 30 : clients MAY cache 30 s -> a record change is invisible to them until expiry
